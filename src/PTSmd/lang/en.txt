# ---------------------------------------------------------------- ENCODE notes: textures
tex.unknown = not a BMP or TGA the game can read - save it as a 24-bit BMP or an uncompressed 24/32-bit TGA
tex.bmp_bits = {0}-bit BMP: the game reads 8, 24 and 32-bit BMPs - save it as 24-bit
tex.bmp_compressed = compressed BMP: the game reads only uncompressed BMPs - save it without compression
tex.bmp_damaged = damaged BMP (the pixels do not fit in the file)
tex.bmp_flipped = top-down BMP turned bottom-up (the game reads bottom-up rows)
tex.bmp_moved = pixels moved right behind the header (the game does not read the pixel offset)
tex.tga_cmap = colour-mapped TGA: the game reads only true-colour TGAs - save it as 24 or 32-bit
tex.tga_bits = {0}-bit TGA: the game reads 16, 24 and 32-bit TGAs
tex.tga_rle_damaged = damaged RLE data in the TGA
tex.tga_rle = RLE-packed TGA unpacked (the game reads only uncompressed TGAs)
tex.tga_damaged = damaged TGA (the pixels do not fit in the file)
tex.tga_id = TGA image ID / colour map left out (the game reads the pixels right after the header)
tex.tga_grey = grey TGA saved as colour (the game reads 16, 24 and 32-bit TGAs)
tex.tga_grey_bits = {0}-bit grey TGA: the game cannot read it - save it as a 24-bit TGA
